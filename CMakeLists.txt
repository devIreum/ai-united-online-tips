cmake_minimum_required(VERSION 3.16)
project(ai_united_online_tips LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
add_library(tips_service cpp/tips_service.cpp)
target_include_directories(tips_service PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cpp)
target_link_libraries(tips_service PUBLIC OpenSSL::Crypto SQLite::SQLite3)
add_executable(tips_demo cpp/demo.cpp)
target_link_libraries(tips_demo PRIVATE tips_service)
add_executable(tips_tests cpp/tests.cpp)
target_link_libraries(tips_tests PRIVATE tips_service)
enable_testing()
add_test(NAME tips_tests COMMAND tips_tests)
